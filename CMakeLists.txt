cmake_minimum_required(VERSION 3.16)
project(stm32_power_fail_safe_event_journal C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(EJ_ENABLE_WERROR "Treat warnings as errors" ON)

add_library(event_journal src/event_journal.c)
target_include_directories(event_journal PUBLIC include)
target_compile_options(event_journal PRIVATE -Wall -Wextra -Wpedantic)
if(EJ_ENABLE_WERROR)
  target_compile_options(event_journal PRIVATE -Werror)
endif()

add_library(fake_nor simulator/fake_nor_flash.c)
target_include_directories(fake_nor PUBLIC include simulator)
target_link_libraries(fake_nor PUBLIC event_journal)
target_compile_options(fake_nor PRIVATE -Wall -Wextra -Wpedantic)
if(EJ_ENABLE_WERROR)
  target_compile_options(fake_nor PRIVATE -Werror)
endif()

enable_testing()
add_executable(test_event_journal tests/test_event_journal.c)
target_include_directories(test_event_journal PRIVATE include simulator)
target_link_libraries(test_event_journal PRIVATE event_journal fake_nor)
target_compile_options(test_event_journal PRIVATE -Wall -Wextra -Wpedantic)
if(EJ_ENABLE_WERROR)
  target_compile_options(test_event_journal PRIVATE -Werror)
endif()
add_test(NAME event_journal_tests COMMAND test_event_journal)

add_executable(journal_demo app/main.c)
target_include_directories(journal_demo PRIVATE include simulator)
target_link_libraries(journal_demo PRIVATE event_journal fake_nor)
target_compile_options(journal_demo PRIVATE -Wall -Wextra -Wpedantic)
if(EJ_ENABLE_WERROR)
  target_compile_options(journal_demo PRIVATE -Werror)
endif()
